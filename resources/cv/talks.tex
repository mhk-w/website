% One \item = one talk. \talktitle prints the title once; each \talkrow
% below it is one independent venue/date instance (a talk given multiple
% times gets multiple \talkrow calls) — every row uses the same macro,
% so there's no special-cased "first row" to keep track of when editing.
\newcommand{\talktitle}[1]{%
  \small\color{maincolor}\textbf{\color{hlcolor-0}{#1}}\\
}

\newcommand{\talkrow}[3]{%
  \small\color{maincolor}%
  \begin{tabular}[t]{p{0.8\linewidth}@{\hspace{0.03\linewidth}}p{0.15\linewidth}}
    \talktag{#1}\ \textit{#3}
    & \raggedleft\textbf{\textcolor{maincolor}{#2}}
  \end{tabular}%
}

\newcommand{\talktag}[1]{%
  \ifstrequal{#1}{Seminar}{\tagSeminar}{%
  \ifstrequal{#1}{Lecture}{\tagLecture}{%
  \ifstrequal{#1}{Exhibition}{\tagExhibition}{%
  \tagPanel}}}%
}
